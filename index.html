WeaponData.cs

Scriptable Object weapon definition — new weapons are created as assets, not new scripts. Used in Space Runaway.

[CreateAssetMenu(fileName = "NewWeapon", menuName = "Combat/Weapon Data")]
public class WeaponData : ScriptableObject
{
    public string weaponName;
    public GameObject projectilePrefab;
    public float fireRate;
    public float damage;
    public float projectileSpeed;

    public void Fire(Transform origin, Vector2 direction)
    {
        var projectile = Instantiate(projectilePrefab, origin.position, Quaternion.identity);
        var rb = projectile.GetComponent<Rigidbody2D>();
        rb.linearVelocity = direction.normalized * projectileSpeed;

        if (projectile.TryGetComponent(out Projectile p))
            p.Init(damage);
    }
}

WaveSpawner.cs

Configurable enemy wave spawner — wave data lives in Scriptable Objects so difficulty tuning doesn't touch code.

public class WaveSpawner : MonoBehaviour
{
    [SerializeField] private WaveData[] waves;
    [SerializeField] private Transform[] spawnPoints;

    private int currentWave = 0;

    public IEnumerator RunWave()
    {
        WaveData wave = waves[currentWave];

        for (int i = 0; i < wave.enemyCount; i++)
        {
            var point = spawnPoints[Random.Range(0, spawnPoints.Length)];
            Instantiate(wave.enemyPrefab, point.position, Quaternion.identity);
            yield return new WaitForSeconds(wave.spawnInterval);
        }

        currentWave = Mathf.Min(currentWave + 1, waves.Length - 1);
    }
}

{
    try
    {
        var result = await auth.SignInWithEmailAndPasswordAsync(email, password);
        return AuthResult.Success(result.User);
    }
    catch (FirebaseException e)
    {
        return AuthResult.Failure(MapAuthError(e));
    }
